// ADLTIX Edge Function: reassign-lead // رفض إعادة تخصيص مؤثرين الأجنت import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; // ---------- CORS ---------- // ملف مشترك بين كل الـ Edge Functions — يسمح للموقع يتصل بيها من غير مشاكل CORS export const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", }; export function jsonResponse(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } export function handleOptions(req: Request): Response | null { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } return null; } // ---------- Session ---------- // ملف مشترك — التحقق من الـ token بتاع الجلسة (بديل sessionOf_ في Apps Script) export function getSupabaseAdmin() { return createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")! ); } export interface SessionInfo { username: string; role: string; staff_name: string; } export async function sessionOf(token: string | undefined): Promise { if (!token) return null; const supabase = getSupabaseAdmin(); const { data, error } = await supabase .from("app_sessions") .select("username, role, staff_name") .eq("token", token) .maybeSingle(); if (error || !data) return null; return data as SessionInfo; } export function isAdminOrManager(session: SessionInfo | null): boolean { return !!session && (session.role === "admin" || session.role === "manager"); } // ---------- الكود ---------- // Edge Function: reassign-lead // // نقل مؤثر موجود لموظفة تانية، بقواعد واضحة: // // • المؤثر محدش كلّمه → أي موظفة تقدر تاخده لنفسها ✅ // • الموظفة الأصلية بعتتله رسالة → مقفول، الموظفة متقدرش تاخده ❌ // • الأدمن/المدير → يقدر ينقله غصب، بس بتسجيل صريح // // كل عملية بتتسجّل في assignment_log بالتاريخ ومين عملها ومين كان صاحبها. Deno.serve(async (req) => { const optionsRes = handleOptions(req); if (optionsRes) return optionsRes; try { const { token, uid, toStaff, force } = await req.json(); const session = await sessionOf(token); if (!session) return jsonResponse({ ok: false, error: "سجّل دخول تاني" }, 401); const role = String(session.role || "").toLowerCase(); const isAdmin = role === "admin" || role === "manager"; const me = session.staff_name || String(session.username || "").split("@")[0]; const cleanUid = String(uid || "").trim().toLowerCase(); if (!cleanUid) return jsonResponse({ ok: false, error: "uid مطلوب" }, 400); // الموظفة بتاخده لنفسها. الأدمن يقدر يحدد لمين. const target = isAdmin ? (String(toStaff || "").trim() || me) : me; const supabase = getSupabaseAdmin(); const { data: row } = await supabase .from("influencers") .select("uid, handle, assigned_to, archived, message_status, signup_status, followup_count") .eq("uid", cleanUid) .maybeSingle(); if (!row) return jsonResponse({ ok: false, error: "المؤثر ده مش موجود" }, 404); // هل اتعمله أي أكشن قبل كده؟ const { count: actionCount } = await supabase .from("actions_log") .select("id", { count: "exact", head: true }) .eq("uid", cleanUid); const hasActions = (actionCount ?? 0) > 0; const statusLocked = ["sent", "replied", "followup", "contacted"].includes(String(row.message_status || "").toLowerCase()) || ["subscribed", "signed_up"].includes(String(row.signup_status || "").toLowerCase()) || (Number(row.followup_count) || 0) > 0; const isLocked = hasActions || statusLocked; const owner = String(row.assigned_to || "").trim(); // هو أصلاً عندها؟ if (owner && owner === target) { return jsonResponse({ ok: true, result: { changed: false, alreadyYours: true, uid: cleanUid, owner, message: "المؤثر ده معاكي أصلاً ✅" }, }); } // 🤖 محجوز للأجنت: عنده إيميل/واتساب، فالأجنت بيكلّمه أوتوماتيك. // الموظفة متقدرش تسحبه — وإلا هيتكلّم مرتين والعميل يشوف ازدواجية. if (!owner && !isAdmin) { return jsonResponse({ ok: false, agentOwned: true, error: "المؤثر ده محجوز للـ Reda Ai Agent — عنده إيميل أو واتساب " + "والأجنت بيكلّمه أوتوماتيك. لو محتاجاه، كلّمي المدير.", }, 409); } // الأدمن بياخده من الأجنت؟ لازم يأكد — عشان ميحصلش تواصل مزدوج if (!owner && isAdmin && force !== true) { return jsonResponse({ ok: false, needsForce: true, agentOwned: true, error: "المؤثر ده محجوز للـ Reda Ai Agent. لو نقلته لموظفة، " + "ممكن يتكلّم مرتين. لو متأكد، أكّد إعادة التخصيص.", }, 409); } // 🤖 محجوز للأجنت: مالوش صاحب لأن عنده إيميل/واتساب. // الموظفة متقدرش تسحبه — الأجنت بيكلّمه أوتوماتيك، ولو اتنقل // لموظفة هيحصل تواصل مزدوج والمؤثر هيستلم رسالتين. if (!owner && !isAdmin) { return jsonResponse({ ok: false, agentOwned: true, error: "المؤثر ده تبع الـ AI Agent — عنده إيميل أو واتساب " + "والأجنت بيكلّمه أوتوماتيك، فمينفعش يتنقل لموظفة.", }, 409); } // 🔒 مقفول: الموظفة الأصلية بعتتله. الموظفة العادية متقدرش تاخده. if (isLocked && !isAdmin) { return jsonResponse({ ok: false, locked: true, owner: owner || null, error: owner ? `المؤثر ده تبع ${owner} وبعتتله بالفعل — مينفعش تتعمله إعادة تخصيص. لو محتاجاه، كلّمي المدير.` : "المؤثر ده اتبعتله رسالة بالفعل — مينفعش تتعمله إعادة تخصيص.", }, 409); } // الأدمن بينقل مقفول؟ لازم يأكد صراحةً if (isLocked && isAdmin && force !== true) { return jsonResponse({ ok: false, needsForce: true, locked: true, owner: owner || null, error: `المؤثر ده ${owner ? "تبع " + owner + " و" : ""}اتبعتله رسالة بالفعل. لو متأكد، أكّد إعادة التخصيص.`, }, 409); } const updates: Record = { assigned_to: target }; if (row.archived === true) updates.archived = false; // لو كان متأرشف، يرجع const { error: updErr } = await supabase .from("influencers").update(updates).eq("uid", cleanUid); if (updErr) return jsonResponse({ ok: false, error: updErr.message }, 500); // 📝 التسجيل — ده اللي المدير هيتابع منه await supabase.from("assignment_log").insert({ uid: cleanUid, handle: row.handle || null, from_staff: owner || null, to_staff: target, by_user: me, by_role: role, reason: row.archived === true ? "restore_archived" : (!owner ? "from_agent" : (isAdmin && isLocked ? "admin_force" : "self_claim")), was_locked: isLocked, note: isLocked && isAdmin ? "نقل إداري رغم وجود تواصل سابق" : null, }); const { data: fresh } = await supabase .from("influencers").select("*").eq("uid", cleanUid).single(); return jsonResponse({ ok: true, result: { changed: true, uid: cleanUid, from: owner || null, to: target, wasLocked: isLocked, card: fresh, message: owner ? `تمت إعادة التخصيص من ${owner} ليك ✅` : "تمت إعادة التخصيص ليك ✅", }, }); } catch (err) { return jsonResponse({ ok: false, error: String(err) }, 500); } });
