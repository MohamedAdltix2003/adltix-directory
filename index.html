// ADLTIX Edge Function: add-employee // 🐞 إصلاح: الموظفة اللي اتحذفت قبل كده كانت بتتضاف في app_users // بس تفضل معطّلة في staff_list — فمش بتدخل التوزيع ولا تظهر في الفريق. import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; // ---------- CORS ---------- // ملف مشترك بين كل الـ Edge Functions — يسمح للموقع يتصل بيها من غير مشاكل CORS export const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", }; export function jsonResponse(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } export function handleOptions(req: Request): Response | null { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } return null; } // ---------- Session ---------- // ملف مشترك — التحقق من الـ token بتاع الجلسة (بديل sessionOf_ في Apps Script) export function getSupabaseAdmin() { return createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")! ); } export interface SessionInfo { username: string; role: string; staff_name: string; } export async function sessionOf(token: string | undefined): Promise { if (!token) return null; const supabase = getSupabaseAdmin(); const { data, error } = await supabase .from("app_sessions") .select("username, role, staff_name") .eq("token", token) .maybeSingle(); if (error || !data) return null; return data as SessionInfo; } export function isAdminOrManager(session: SessionInfo | null): boolean { return !!session && (session.role === "admin" || session.role === "manager"); } // ---------- الكود ---------- // Edge Function: add-employee // بديل apiAddEmployee_ — أدمن أو مدير بس يقدر يضيف موظفة جديدة Deno.serve(async (req) => { const optionsRes = handleOptions(req); if (optionsRes) return optionsRes; try { const { token, username, password, staffName, emoji } = await req.json(); const session = await sessionOf(token); if (!isAdminOrManager(session)) { return jsonResponse({ ok: false, error: "مش مسموح — أدمن/مدير بس" }, 403); } if (!username || !password || !staffName) { return jsonResponse({ ok: false, error: "بيانات ناقصة" }, 400); } const supabase = getSupabaseAdmin(); // تأكد إن اليوزرنيم مش مكرر const { data: existing } = await supabase .from("app_users") .select("username") .ilike("username", username.trim()) .maybeSingle(); if (existing) { return jsonResponse({ ok: false, error: "اليوزرنيم ده مستخدم" }, 409); } const { error: insertUserErr } = await supabase.from("app_users").insert({ username: username.trim(), password, role: "staff", staff_name: staffName, active: true, }); if (insertUserErr) { return jsonResponse({ ok: false, error: insertUserErr.message }, 500); } // staff_list: نضيف أو **نعيد التفعيل** // // 🐞 البق اللي كان هنا: حذف الموظفة بيعطّلها (active=false) مش بيمسحها، // عشان سجل الأكشنات يفضل مربوط باسمها. لكن الكود القديم كان بيشوف // الصف موجود فيتخطى الإضافة — و active يفضل false. // النتيجة: الموظفة بتتضاف في app_users بس مش داخلة التوزيع // ولا ظاهرة في إدارة الفريق. دلوقتي بنعيد تفعيلها صراحةً. const { data: staffExists } = await supabase .from("staff_list") .select("staff_name, active") .eq("staff_name", staffName) .maybeSingle(); let reactivated = false; if (!staffExists) { await supabase.from("staff_list").insert({ staff_name: staffName, emoji: emoji || "👤", active: true, }); } else { reactivated = staffExists.active === false; await supabase .from("staff_list") .update({ active: true, emoji: emoji || "👤" }) .eq("staff_name", staffName); } // ⚠️ قبل كده الرسالة كانت بتقول "تم التوزيع" من غير ما يحصل أي توزيع فعلي. // دلوقتي بنعيد التوزيع بجد عشان الموظفة الجديدة تاخد نصيبها. // rebalance_leads بتحمي المؤثرين اللي اتبعتلهم رسايل — دول بيفضلوا مكانهم. let redistributed = 0; let distributionOk = false; const staffLeadCount: Record = {}; try { const { data: rb, error: rbErr } = await supabase.rpc("rebalance_leads"); if (!rbErr && rb) { distributionOk = true; for (const r of rb as any[]) { staffLeadCount[r.staff_name] = Number(r.total_now) || 0; redistributed += Number(r.received) || 0; } } } catch (_) { // مبدأ المشروع: الإضافة لازم تنجح دايمًا حتى لو التوزيع فشل } // عدد مؤثرين الأجنت — بيتعرض في الرسالة عشان يبان إنه ثابت // ومش بيتأثر بإضافة موظفة. من غيره المدير بيحس إن الأرقام ناقصة. const { count: agentCount } = await supabase .from("influencers") .select("uid", { count: "exact", head: true }) .eq("archived", false) .is("assigned_to", null); const staffTotal = Object.values(staffLeadCount).reduce((a, b) => a + b, 0); return jsonResponse({ ok: true, result: { added: true, staffName, distributionOk, redistributed, staffLeadCount, reactivated, // كانت موجودة ومعطّلة، ورجّعناها agentCount: agentCount ?? 0, staffTotal, totalActive: staffTotal + (agentCount ?? 0), }, }); } catch (err) { return jsonResponse({ ok: false, error: String(err) }, 500); } });
