-- ========================================================================= -- ADLTIX — 74_PENDING_PROOFS -- الإثباتات اللي الـ AI مقدرش يربطها. -- -- 📌 الفكرة: الموظفة ترفع وخلاص — مفيش وقفة. -- اللي اتربط يتسجّل عادي، واللي فشل يستنى في الداشبورد -- بمقترحات من المؤثرين اللي ضافتهم نفس الوقت. -- ========================================================================= create table if not exists public.pending_proofs ( id bigserial primary key, staff text not null, url text not null, hash text, ai_name text, -- اللي الموديل قراه ai_platform text, ocr_text text, action_type text default 'sent', resolved boolean default false, resolved_uid text, resolved_by text, resolved_at timestamptz, ts timestamptz default now() ); create index if not exists idx_pp_open on public.pending_proofs (staff, ts desc) where not resolved; create unique index if not exists idx_pp_hash on public.pending_proofs (hash) where hash is not null; -- ------------------------------------------------------------------------- -- ➕ تسجيل إثبات معلّق -- ------------------------------------------------------------------------- create or replace function public.add_pending_proof( p_staff text, p_url text, p_hash text default null, p_name text default null, p_platform text default null, p_ocr text default null, p_type text default 'sent' ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_id bigint; begin -- ⚠️ نفس الصورة اترفعت قبل كده؟ if p_hash is not null and exists ( select 1 from public.pending_proofs where hash = p_hash ) then return jsonb_build_object('ok', false, 'duplicate', true, 'message', 'الصورة دي مرفوعة قبل كده'); end if; insert into public.pending_proofs (staff, url, hash, ai_name, ai_platform, ocr_text, action_type) values (p_staff, p_url, nullif(p_hash,''), nullif(p_name,''), nullif(p_platform,''), left(coalesce(p_ocr,''), 400), coalesce(nullif(p_type,''), 'sent')) returning id into v_id; return jsonb_build_object('ok', true, 'id', v_id); end; $$; -- ------------------------------------------------------------------------- -- 📋 للداشبورد — مع مقترحات لكل صورة -- -- 🤖 المقترحات: المؤثرين اللي الموظفة ضافتهم في نفس الفترة، -- مرتّبين بالأقرب زمنيًا لوقت الصورة. -- ------------------------------------------------------------------------- create or replace function public.pending_overview(p_limit int default 200) returns jsonb language plpgsql stable as $$ declare v jsonb; begin select coalesce(jsonb_agg(x order by (x->>'ts') desc), '[]'::jsonb) into v from ( select jsonb_build_object( 'id', p.id, 'staff', p.staff, 'url', p.url, 'ai_name', coalesce(p.ai_name, ''), 'platform', coalesce(p.ai_platform, ''), 'ocr', coalesce(p.ocr_text, ''), 'type', p.action_type, 'ts', p.ts, -- 💡 المقترحين — اللي ضافتهم قريب من وقت الصورة 'suggest', coalesce(( select jsonb_agg(jsonb_build_object( 'uid', s.uid, 'handle', s.handle, 'display', coalesce(s.display_name, ''), 'platform', s.platform, 'mins', s.mins ) order by s.mins) from ( select i.uid, replace(coalesce(i.handle,''),'@','') as handle, i.display_name, i.platform, abs(extract(epoch from (i.imported_at - p.ts)) / 60)::int as mins from public.influencers i where i.assigned_to = p.staff and coalesce(i.archived,false) = false and i.outcome_type is null and lower(coalesce(i.message_status,'new')) in ('new','') and i.imported_at between p.ts - interval '6 hours' and p.ts + interval '2 hours' order by abs(extract(epoch from (i.imported_at - p.ts))) limit 5 ) s ), '[]'::jsonb) ) as x from public.pending_proofs p where not p.resolved order by p.ts desc limit p_limit ) t; return jsonb_build_object( 'items', v, 'total', (select count(*) from public.pending_proofs where not resolved), 'by_staff', coalesce(( select jsonb_agg(jsonb_build_object('staff', s.staff, 'count', s.n) order by s.n desc) from (select staff, count(*) n from public.pending_proofs where not resolved group by staff) s ), '[]'::jsonb) ); end; $$; -- ------------------------------------------------------------------------- -- ✅ ربط الإثبات بمؤثر — بيسجّل الأكشن كامل -- ------------------------------------------------------------------------- create or replace function public.resolve_pending( p_id bigint, p_uid text, p_by text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare p record; v_inf record; v_seq int; v_type text; begin select * into p from public.pending_proofs where id = p_id and not resolved; if p is null then return jsonb_build_object('ok', false, 'error', 'الإثبات مش موجود'); end if; select * into v_inf from public.influencers where uid = p_uid; if v_inf is null then return jsonb_build_object('ok', false, 'error', 'المؤثر مش موجود'); end if; -- الأكشن: إرسال ولا متابعة؟ if lower(coalesce(v_inf.message_status,'new')) in ('new','') then v_type := 'sent'; v_seq := 1; else v_type := 'followup'; v_seq := coalesce(v_inf.followup_count,0) + 1; end if; insert into public.actions_log (uid, staff, action_type, seq, screenshot_url, source, ts) values (p_uid, p.staff, v_type, v_seq, p.url, 'dashboard', p.ts); -- البصمة if p.hash is not null then insert into public.proof_hashes (hash, uid, staff, action_type, seq, url, confidence, ts) values (p.hash, p_uid, p.staff, v_type, v_seq, p.url, 'manual', p.ts) on conflict (hash) do nothing; end if; -- حالة المؤثر if v_type = 'sent' then update public.influencers set message_status = 'sent', sent_at = coalesce(sent_at, p.ts) where uid = p_uid; else update public.influencers set followup_count = v_seq, last_followup_at = p.ts, message_status = 'followup' where uid = p_uid; end if; update public.pending_proofs set resolved = true, resolved_uid = p_uid, resolved_by = p_by, resolved_at = now() where id = p_id; return jsonb_build_object('ok', true, 'type', v_type, 'seq', v_seq, 'handle', v_inf.handle); end; $$; select 'جاهز' as status, (select count(*) from public.pending_proofs where not resolved) as معلّقة;
